/* TogetherLogo — reusable brand mark for the "Together" couple-organizer app.
   Plain-JS React component (reads window.React at call time so the host controls
   when React is present). No JSX so no transpile-time React global is assumed. */

export function TogetherLogo(props) {
  var React = window.React;
  props = props || {};

  var height    = props.height != null ? props.height : 64;
  var variant   = props.variant || "horizontal";
  var word      = props.word != null ? props.word : "Together";
  var tagline   = props.tagline || "";
  var colorA    = props.colorA || "var(--haushalt)";
  var colorB    = props.colorB || "var(--finanzen)";
  var wordColor = props.wordColor || "var(--text)";
  var blendMode = props.blendMode || "multiply"; // "multiply" on light, "screen" on dark

  var h = height;
  var w = Math.round(h * (120 / 90)); // svg aspect ratio

  var mark = React.createElement(
    "svg",
    {
      width: w,
      height: h,
      viewBox: "0 0 120 90",
      role: "img",
      "aria-label": word ? word + " Logo" : "Together Logo"
    },
    React.createElement(
      "g",
      { style: { mixBlendMode: blendMode } },
      React.createElement("circle", { cx: 46, cy: 45, r: 33, style: { fill: colorA } }),
      React.createElement("circle", { cx: 74, cy: 45, r: 33, style: { fill: colorB } })
    )
  );

  if (variant === "mark" || !word) {
    return mark;
  }

  var wordSize = Math.round(h * 0.62);
  var tagSize  = Math.max(11, Math.round(h * 0.2));

  var textBlock = React.createElement(
    "div",
    {
      style: {
        display: "flex",
        flexDirection: "column",
        gap: Math.round(h * 0.06) + "px",
        lineHeight: 1
      }
    },
    React.createElement(
      "span",
      {
        style: {
          fontFamily: "var(--font-headline), 'Fredoka', system-ui, sans-serif",
          fontWeight: 600,
          fontSize: wordSize + "px",
          letterSpacing: "-0.5px",
          color: wordColor
        }
      },
      word
    ),
    tagline
      ? React.createElement(
          "span",
          {
            style: {
              fontFamily: "var(--font-body), 'Nunito', system-ui, sans-serif",
              fontWeight: 600,
              fontSize: tagSize + "px",
              color: "var(--text-secondary)"
            }
          },
          tagline
        )
      : null
  );

  return React.createElement(
    "div",
    {
      style: {
        display: "inline-flex",
        flexDirection: variant === "vertical" ? "column" : "row",
        alignItems: "center",
        gap: Math.round(h * (variant === "vertical" ? 0.28 : 0.3)) + "px",
        textAlign: variant === "vertical" ? "center" : "left"
      }
    },
    mark,
    textBlock
  );
}
